  Add this NEW endpoint:

import DailySummary from "../models/DailySummary.js";

/**
 * GET /api/finance/summary/today
 * Fetch pre-aggregated daily summary (ultra-fast)
 * Response time: 1-5ms
 */
router.get("/summary/today", adminAuth, async (req, res) => {
  try {
    const today = new Date();
    today.setHours(0, 0, 0, 0);

    const summary = await DailySummary.findOne({ date: today });

    if (!summary) {
      return res.json({
        totalRevenue: 0,
        newJoiningRevenue: 0,
        renewalRevenue: 0,
        totalTransactions: 0,
        incomeByPlan: {},
        incomeByTrainingType: {},
        membersByTrainingType: {},
        lastUpdatedAt: new Date(),
      });
    }

    res.json({
      totalRevenue: summary.totalRevenue,
      newJoiningRevenue: summary.newJoiningRevenue,
      renewalRevenue: summary.renewalRevenue,
      totalTransactions: summary.totalTransactions,
      incomeByPlan: Object.fromEntries(summary.incomeByPlan || new Map()),
      incomeByTrainingType: Object.fromEntries(summary.incomeByTrainingType || new Map()),
      membersByTrainingType: Object.fromEntries(summary.membersByTrainingType || new Map()),
      lastUpdatedAt: summary.lastUpdatedAt,
    });
  } catch (err) {
    console.error("Summary error:", err);
    res.status(500).json({ message: "Failed to fetch summary", error: err.message });
  }
});

  Verify:
    $ curl http://localhost:5000/api/finance/summary/today

Task 4.2: Add Pagination to Transactions Endpoint

/**
 * GET /api/finance/transactions/:gymId?page=1&limit=20
 * Fetch paginated transaction list
 */
router.get("/transactions/:gymId", adminAuth, async (req, res) => {
  try {
    const { gymId } = req.params;
    const page = Math.max(1, parseInt(req.query.page) || 1);
    const limit = Math.min(50, parseInt(req.query.limit) || 20);
    const skip = (page - 1) * limit;

    const total = await FinanceLog.countDocuments({ 
      gymId: gymId 
    });

    const transactions = await FinanceLog.find({ 
      gymId: gymId 
    })
      .sort({ date: -1 })
      .skip(skip)
      .limit(limit)
      .lean();

    res.json({
      data: transactions,
      pagination: {
        page,
        limit,
        total,
        pages: Math.ceil(total / limit),
      },
    });
  } catch (err) {
    console.error("Transaction fetch error:", err);
    res.status(500).json({ message: "Failed to fetch transactions" });
  }
});
