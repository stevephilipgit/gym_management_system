ANALYSIS OF CURRENT QUERIES:

1. Dashboard Aggregations:
   FinanceLog.find({date: {$gte: start, $lte: end}})
   GROUP BY: type, trainingType, plan
   PROBLEM: Full table scan without date index

2. Member Registration:
   Member.find({createdAt: {$gte: start, $lte: end}})
   PROBLEM: Full collection scan without timestamp index

3. Payment History:
   PaymentLog.find({gymId: gymId})
   PROBLEM: No index on foreign key lookup

4. Transaction Details:
   FinanceLog.find({gymId: gymId}, {_id, date, amount})
   PROBLEM: Compound query without optimization

```javascript
import mongoose from "mongoose";

const dailySummarySchema = new mongoose.Schema(
  {
    // Compound unique key: date + gym (for multi-gym future)
    date: {
      type: Date,
      required: true,
      index: true,
      set: (v) => {
        // Store only date part (00:00:00), ignore time
        const d = new Date(v);
        d.setHours(0, 0, 0, 0);
        return d;
      },
    },

    // SUMMARY TOTALS
    totalRevenue: {
      type: Number,
      default: 0,
      min: 0,
    },

    newJoiningRevenue: {
      type: Number,
      default: 0,
      min: 0,
    },

    renewalRevenue: {
      type: Number,
      default: 0,
      min: 0,
    },

    totalTransactions: {
      type: Number,
      default: 0,
      min: 0,
    },

    // BREAKDOWNS (Nested objects for structure)
    incomeByPlan: {
      type: Map,
      of: Number,
      default: new Map(),
      // Example: {"1 Month": 5000, "3 Months": 10000}
    },

    incomeByTrainingType: {
      type: Map,
      of: Number,
      default: new Map(),
      // Example: {"Weight Loss": 8000, "Weight Gain": 7000}
    },

    membersByTrainingType: {
      type: Map,
      of: Number,
      default: new Map(),
      // Example: {"Weight Loss": 3, "Weight Gain": 2}
    },

    // METADATA
    lastUpdatedAt: {
      type: Date,
      default: Date.now,
      index: true,
    },

    isCompleted: {
      type: Boolean,
      default: false,
      // True if date is in past (data immutable)
    },
  },
  { 
    timestamps: true,
    // Unique index: only one summary per date
    indexes: [{ fields: { date: 1 }, unique: true }]
  }
);

// Compound index for efficient lookups
dailySummarySchema.index({ date: 1, isCompleted: 1 });

export default mongoose.model("DailySummary", dailySummarySchema);
```

DETAILED SCHEMA BREAKDOWN:

Field               Type           Purpose
─────────────────────────────────────────────────────
date                Date (00:00)   Key for the day (indexed, unique)
totalRevenue        Number         SUM(amount) for the day
newJoiningRevenue   Number         SUM(amount) WHERE type="new"
renewalRevenue      Number         SUM(amount) WHERE type="renew"
totalTransactions   Number         COUNT(*) transactions
incomeByPlan        Map<String>    {plan: amount} breakdown
incomeByTrainingType Map<String>   {trainingType: amount}
membersByTrainingType Map<String>  {trainingType: count}
lastUpdatedAt       Date (time)    Last update timestamp
isCompleted         Boolean        true if date < today

Why Map instead of nested object?
  • Maps in MongoDB are efficient for key-value lookups
  • Flexible schema (new plans don't need migration)
  • Direct query: db.dailysummary.find({"incomeByPlan.1 Month": {$gt: 0}})

Why separate lastUpdatedAt?
  • Can rebuild today's summary without touching historical data
  • Useful for debugging: "summary last updated at 3:45 PM"

Why isCompleted flag?
  • Today's record is mutable (constantly updated)
  • Yesterday's record is immutable (locking)
  • At midnight: new record created, old record marked complete
  • Prevents accidental changes to past data

```javascript
import DailySummary from "../models/DailySummary.js";
import FinanceLog from "../models/FinanceLog.js";

/**
 * Gets today's summary, creating if not exists
 * Called whenever a transaction is recorded
 */
export async function getTodaySummary() {
  const now = new Date();
  const today = new Date(now);
  today.setHours(0, 0, 0, 0); // Start of day

  // Try to find existing summary
  let summary = await DailySummary.findOne({ date: today });

  // If not found, create new summary for today
  if (!summary) {
    summary = new DailySummary({
      date: today,
      totalRevenue: 0,
      newJoiningRevenue: 0,
      renewalRevenue: 0,
      totalTransactions: 0,
      incomeByPlan: new Map(),
      incomeByTrainingType: new Map(),
      membersByTrainingType: new Map(),
    });
    await summary.save();
  }

  return summary;
}

/**
 * Updates today's summary with a new transaction
 * Incremental update (not full recalculation)
 */
export async function updateTodaySummary(transactionLog) {
  const summary = await getTodaySummary();

  const amount = Number(transactionLog.amount) || 0;
  const plan = transactionLog.plan || "Unknown";
  const trainingType = transactionLog.trainingType || "Unknown";
  const type = transactionLog.type || "new"; // "new" or "renew"

  // Atomic increment operations
  const updateOps = {
    $inc: {
      totalRevenue: amount,
      totalTransactions: 1,
    },
    $set: {
      lastUpdatedAt: new Date(),
    },
  };

  // Add to "new" or "renewal" column
  if (type === "new") {
    updateOps.$inc.newJoiningRevenue = amount;
  } else if (type === "renew") {
    updateOps.$inc.renewalRevenue = amount;
  }

  // Increment plan breakdown (CRITICAL: use $inc for map values)
  updateOps.$inc[`incomeByPlan.${plan}`] = amount;
  updateOps.$inc[`incomeByTrainingType.${trainingType}`] = amount;

  // Update summary
  await DailySummary.findByIdAndUpdate(
    summary._id,
    updateOps,
    { new: true }
  );
}

/**
 * Marks yesterday's summary as completed (immutable)
 * Called at midnight via cron job or app startup check
 */
export async function markPreviousDayComplete() {
  const yesterday = new Date();
  yesterday.setDate(yesterday.getDate() - 1);
  yesterday.setHours(0, 0, 0, 0);

  await DailySummary.findOneAndUpdate(
    { date: yesterday },
    { isCompleted: true },
    { upsert: false }
  );
}

/**
 * Recalculates today's summary from scratch
 * Use if summary gets corrupted
 */
export async function rebuildTodaySummary() {
  const today = new Date();
  today.setHours(0, 0, 0, 0);

  const tomorrow = new Date(today);
  tomorrow.setDate(tomorrow.getDate() + 1);

  // Fetch all transactions for today
  const transactions = await FinanceLog.find({
    date: { $gte: today, $lt: tomorrow },
  });

  // Initialize
  let totalRevenue = 0;
  let newRevenue = 0;
  let renewalRevenue = 0;
  const incomeByPlan = new Map();
  const incomeByTrainingType = new Map();
  const membersByTrainingType = new Map();

  // Loop through transactions to aggregate
  for (const tx of transactions) {
    const amount = Number(tx.amount) || 0;
    totalRevenue += amount;

    if (tx.type === "new") newRevenue += amount;
    if (tx.type === "renew") renewalRevenue += amount;

    // Plan breakdown
    const plan = tx.plan || "Unknown";
    incomeByPlan.set(plan, (incomeByPlan.get(plan) || 0) + amount);

    // Training type breakdown
    const trainingType = tx.trainingType || "Unknown";
    incomeByTrainingType.set(
      trainingType,
      (incomeByTrainingType.get(trainingType) || 0) + amount
    );
  }

  // Get member count by training type for today
  const Member = require("../models/Member.js").default;
  const memberCountAgg = await Member.aggregate([
    {
      $match: {
        paymentStatus: "paid",
        createdAt: { $gte: today, $lt: tomorrow },
      },
    },
    {
      $group: {
        _id: "$trainingType",
        count: { $sum: 1 },
      },
    },
  ]);

  memberCountAgg.forEach((m) => {
    membersByTrainingType.set(m._id || "Unknown", m.count);
  });

  // Upsert summary
  await DailySummary.findOneAndUpdate(
    { date: today },
    {
      totalRevenue,
      newJoiningRevenue: newRevenue,
      renewalRevenue,
      totalTransactions: transactions.length,
      incomeByPlan,
      incomeByTrainingType,
      membersByTrainingType,
      lastUpdatedAt: new Date(),
    },
    { upsert: true, new: true }
  );
}
```

```javascript
// Check if new day and mark yesterday as complete
async function dailyMidnightTask() {
  const now = new Date();
  const lastCheck = new Date(); // Track last execution

  const checkInterval = setInterval(async () => {
    const current = new Date();
    
    // If day changed, execute once
    if (current.getDate() !== lastCheck.getDate()) {
      try {
        console.log("🌙 Midnight task: Marking yesterday complete");
        await markPreviousDayComplete();
        lastCheck.setTime(current.getTime());
      } catch (err) {
        console.error("Midnight task error:", err);
      }
    }
  }, 60000); // Check every minute

  return () => clearInterval(checkInterval);
}

// Start on server init
dailyMidnightTask();
```

```javascript
// Memory cache (no Redis needed for <100K/day)
class SimpleCache {
  constructor() {
    this.cache = new Map();
  }

  set(key, value, ttl = 60000) {
    // ttl: time to live in milliseconds
    this.cache.set(key, { value, expires: Date.now() + ttl });
    setTimeout(() => this.cache.delete(key), ttl);
  }

  get(key) {
    const item = this.cache.get(key);
    if (!item) return null;
    if (item.expires < Date.now()) {
      this.cache.delete(key);
      return null;
    }
    return item.value;
  }

  clear(pattern) {
    // Clear all keys matching pattern
    for (const [key] of this.cache) {
      if (key.includes(pattern)) {
        this.cache.delete(key);
      }
    }
  }
}

export default new SimpleCache();
```

```javascript
// Get gym's timezone (could be stored in settings)
const GYM_TIMEZONE = process.env.GYM_TIMEZONE || "Asia/Kolkata";

export function getTodayRange() {
  // Install: npm install date-fns-tz
  // const { startOfDay, endOfDay } = require("date-fns-tz");
  
  const now = new Date();
  const today = startOfDay(now, { timeZone: GYM_TIMEZONE });
  const tomorrow = endOfDay(now, { timeZone: GYM_TIMEZONE });
  
  return { today, tomorrow };
}

export function getBusinessDayRange(date) {
  // If before 4 AM, belongs to yesterday
  const localHour = new Date(date).toLocaleString('en-US', {
    timeZone: GYM_TIMEZONE,
    hour: '2-digit',
    hour12: false
  });
  
  if (parseInt(localHour) < 4) {
    date.setDate(date.getDate() - 1);
  }
  
  return startOfDay(date, { timeZone: GYM_TIMEZONE });
}
```
