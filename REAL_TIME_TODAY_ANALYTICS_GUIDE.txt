ADD BEFORE the existing /income endpoint:

```javascript
/* ============================================================
   TODAY'S ANALYTICS ENDPOINT (NEW)
============================================================ */
router.get("/today", adminAuth, financeLimiter, async (req, res) => {
  try {
    // Get current server date (or timezone-aware date)
    const now = new Date();
    
    // Business hours consideration:
    // If it's between 12:00 AM - 3:59 AM, consider it "yesterday"
    // This treats the day as: 4:00 AM to 11:00 PM the next day
    let todayStart = new Date(now);
    let todayEnd = new Date(now);
    
    // Set to 4:00 AM (start of business day)
    if (now.getHours() < 4) {
      // Before 4 AM = still yesterday's business day
      todayStart.setDate(todayStart.getDate() - 1);
      todayEnd.setDate(todayEnd.getDate() - 1);
    }
    
    todayStart.setHours(4, 0, 0, 0);
    todayEnd.setHours(23, 59, 59, 999);

    console.log(`📊 Today Analytics Query: ${todayStart} to ${todayEnd}`);

    /* -----------------------------------------
       FETCH TODAY'S FINANCE LOGS
    ----------------------------------------- */
    const logs = await FinanceLog.find({
      date: {
        $gte: todayStart,
        $lte: todayEnd,
      },
    }).sort({ date: -1 });

    /* -----------------------------------------
       MEMBER COUNT BY TRAINING TYPE (Today)
       - Only paid members registered today
    ----------------------------------------- */
    const memberCountsAgg = await Member.aggregate([
      {
        $match: {
          paymentStatus: "paid",
          createdAt: {
            $gte: todayStart,
            $lte: todayEnd,
          },
        },
      },
      {
        $group: {
          _id: "$trainingType",
          count: { $sum: 1 },
        },
      },
    ]);

    const memberCountsByTraining = {};
    memberCountsAgg.forEach((m) => {
      memberCountsByTraining[m._id || "Unknown"] = m.count;
    });

    /* -----------------------------------------
       EMPTY BUT VALID RESPONSE
    ----------------------------------------- */
    if (logs.length === 0) {
      return res.json({
        totalAmount: 0,
        plans: {},
        trainingTypes: {},
        newVsRenew: { new: 0, renewal: 0 },
        logs: [],
        memberCountsByTraining,
        timestamp: new Date(),
        period: "today",
      });
    }

    /* -----------------------------------------
       AGGREGATION (Same as historical endpoint)
    ----------------------------------------- */
    let totalAmount = 0;
    const plans = {};
    const trainingTypes = {};
    const newVsRenew = { new: 0, renewal: 0 };
    const dailyLogs = [];

    for (const log of logs) {
      const amount = Number(log.amount) || 0;
      const plan = normalizePlan(log.plan);
      const training = log.trainingType || "Unknown";

      totalAmount += amount;

      plans[plan] = (plans[plan] || 0) + amount;
      trainingTypes[training] = (trainingTypes[training] || 0) + amount;

      if (log.type === "new") newVsRenew.new += amount;
      if (log.type === "renew") newVsRenew.renewal += amount;

      dailyLogs.push({
        paidAt: log.date.toISOString(),
        amount,
      });
    }

    /* -----------------------------------------
       FINAL RESPONSE
    ----------------------------------------- */
    res.json({
      totalAmount,
      plans,
      trainingTypes,
      newVsRenew,
      logs: dailyLogs,
      memberCountsByTraining,
      timestamp: new Date(),
      period: "today",
    });
  } catch (err) {
    console.error("Today Analytics Error:", err);
    res.status(500).json({ message: "Server Error", error: err.message });
  }
});
```

// Load today's data on mount:
useEffect(() => {
  fetchTodayData();
  
  // Set up auto-refresh every 30 seconds
  const interval = setInterval(() => {
    fetchTodayData();
  }, 30000);
  
  return () => clearInterval(interval);
}, []);

// Midnight reset check:
useEffect(() => {
  const checkMidnight = () => {
    const now = new Date();
    const lastCheck = new Date(lastMidnightCheck);
    
    // If day changed, refresh today's data
    if (now.getDate() !== lastCheck.getDate()) {
      console.log("🌙 New day detected, refreshing today's data");
      fetchTodayData();
      setLastMidnightCheck(now);
    }
  };
  
  const midnightCheck = setInterval(checkMidnight, 60000); // Check every minute
  
  return () => clearInterval(midnightCheck);
}, [lastMidnightCheck]);

// Fetch today's data:
const fetchTodayData = async () => {
  setTodayLoading(true);
  try {
    const res = await axios.get(
      "http://localhost:5000/api/finance/today",
      { withCredentials: true }
    );
    setTodayData(res.data);
    console.log("✅ Today's data refreshed", res.data);
  } catch (err) {
    console.error("Error fetching today's data:", err);
  }
  setTodayLoading(false);
};

STEP 6: REAL-TIME POLLING STRATEGY
================================================================================

Option 1: POLLING (Current recommendation - Simple, Production-ready)
- Frontend calls /today every 30 seconds
- No server-side state needed
- Scalable (no websockets)
- ~2 KB payload every 30 sec

```javascript
const interval = setInterval(fetchTodayData, 30000); // Every 30 seconds
```

Option 2: WEBSOCKETS (Future enhancement)
- Real-time push updates
- Lower latency
- Requires Socket.io setup
- More complex deployment

Option 3: SERVER-SENT EVENTS (Middle ground)
- One-way push from server
- Simpler than websockets
- Doesn't work well with rate limits

RECOMMENDED: Stick with polling (Option 1) for initial release.

STEP 7: BUSINESS HOURS LOGIC (4 AM - 11 PM)
================================================================================

Current implementation considers:
- Before 4:00 AM = Still "yesterday's" business day
- 4:00 AM to 11:59 PM = Current business day
- At midnight (12:00 AM), automatically switches to previous day

Why this works:
- Business opens at 4 AM
- Data from 12:01-3:59 AM belongs to previous day's records
- Midnight refresh handled by frontend check
